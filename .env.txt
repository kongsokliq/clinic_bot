# 🔧 បង្កើតតារាងទិន្នន័យ
def init_db():
    conn = sqlite3.connect('clinic_db.db')
    cursor = conn.cursor()

    # តារាងអ្នកជម្ងឺ
    cursor.execute('''
    CREATE TABLE IF NOT EXISTS patients (
        id INTEGER PRIMARY KEY AUTOINCREMENT,
        name TEXT NOT NULL,
        age INTEGER,
        gender TEXT,
        contact TEXT,
        address TEXT,
        created_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP
    )
    ''')

    # តារាងឯកសារព្យាបាល និងលទ្ធផល
    cursor.execute('''
    CREATE TABLE IF NOT EXISTS medical_records (
        id INTEGER PRIMARY KEY AUTOINCREMENT,
        patient_id INTEGER NOT NULL,
        diagnosis TEXT,
        treatment TEXT,
        result TEXT,
        note TEXT,
        record_date TIMESTAMP DEFAULT CURRENT_TIMESTAMP,
        FOREIGN KEY (patient_id) REFERENCES patients (id)
    )
    ''')

    # តារាងការណាត់ជួប
    cursor.execute('''
    CREATE TABLE IF NOT EXISTS appointments (
        id INTEGER PRIMARY KEY AUTOINCREMENT,
        patient_id INTEGER NOT NULL,
        appointment_date TIMESTAMP NOT NULL,
        purpose TEXT,
        status TEXT DEFAULT 'រង់ចាំ', -- រង់ចាំ / បានជួប / លុបចោល
        created_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP,
        FOREIGN KEY (patient_id) REFERENCES patients (id)
    )
    ''')

    conn.commit()
    conn.close()

# ➕ បន្ថែមអ្នកជម្ងឺ (ឧទាហរណ៍ជាជំហានៗ)
async def add_patient(update: Update, context: ContextTypes.DEFAULT_TYPE):
    query = update.callback_query
    await query.answer()
    await query.edit_message_text("សូមបញ្ចូលឈ្មោះអ្នកជម្ងឺ៖\n(ឧទាហរណ៍: ឈ្មោះ: សុខា អាយុ: 30 ភេទ: ប្រុស ទំនាក់ទំនង: 012345678 អាសយដ្ឋាន: ភ្នំពេញ)")
    context.user_data['action'] = 'add_patient'

async def handle_message(update: Update, context: ContextTypes.DEFAULT_TYPE):
    text = update.message.text
    action = context.user_data.get('action')

    if action == 'add_patient':
        try:
            # វិភាគទិន្នន័យដែលបានបញ្ចូល
            parts = dict(part.split(": ") for part in text.split(" "))
            name = parts.get("ឈ្មោះ", "")
            age = int(parts.get("អាយុ", 0))
            gender = parts.get("ភេទ", "")
            contact = parts.get("ទំនាក់ទំនង", "")
            address = parts.get("អាសយដ្ឋាន", "")

            conn = sqlite3.connect('clinic_db.db')
            cursor = conn.cursor()
            cursor.execute('''
                INSERT INTO patients (name, age, gender, contact, address)
                VALUES (?, ?, ?, ?, ?)
            ''', (name, age, gender, contact, address))
            conn.commit()
            conn.close()

            await update.message.reply_text(f"✅ បានបន្ថែមអ្នកជម្ងឺ **{name}** ដោយជោគជ័យ!")
            context.user_data['action'] = None
        except Exception as e:
            await update.message.reply_text("❌ ទម្រង់ទិន្នន័យមិនត្រឹមត្រូវ សូមព្យាយាមម្តងទៀត។")

def main():
    init_db()
    application = ApplicationBuilder().token(TOKEN).build()

    # ពាក្យបញ្ជាមូលដ្ឋាន
    application.add_handler(CommandHandler("start", start))

    # ប៊ូតុងចុច
    application.add_handler(CallbackQueryHandler(manage_patients, pattern='^manage_patients$'))
    application.add_handler(CallbackQueryHandler(manage_records, pattern='^manage_records$'))
    application.add_handler(CallbackQueryHandler(manage_appointments, pattern='^manage_appointments$'))
    application.add_handler(CallbackQueryHandler(search_data, pattern='^search_data$'))
    application.add_handler(CallbackQueryHandler(back_main, pattern='^back_main$'))
    application.add_handler(CallbackQueryHandler(add_patient, pattern='^add_patient$'))

    # ដោះស្រាយសារអត្ថបទ
    application.add_handler(MessageHandler(filters.TEXT & ~filters.COMMAND, handle_message))

    # ដំណើរការ Bot
    application.run_polling()

if __name__ == "__main__":
